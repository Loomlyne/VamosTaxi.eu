import { test, expect } from "@playwright/test";
import { spawn, type ChildProcess } from "node:child_process";
import { readFileSync } from "node:fs";
import { NEXT_BIN, waitForNextServer, WEB_ROOT } from "../support/server-harness";
import { openInLocale, type Lang } from "../support/locale";
const Q = "/Users/koss/Developer/vamos-wt/class-photo-small/.planning/quick/260930-cps-class-photo-small";
const CLS = [
  { slug: "economy", name: "Economy", pax: 3, bags: 3, t: 11100 },
  { slug: "business", name: "Business", pax: 7, bags: 6, t: 22200 },
  { slug: "van-luxury", name: "Van luxury", pax: 12, bags: 9, t: 33300 },
];
let dev: ChildProcess | null = null;
const base = "http://127.0.0.1:4378";
test.describe.configure({ mode: "serial" });
test.beforeAll(async ({}, ti) => {
  if (ti.project.name !== "component-1440") return;
  ti.setTimeout(200_000);
  dev = spawn(NEXT_BIN, ["dev", "-p", "4378"], { cwd: WEB_ROOT, stdio: "ignore", detached: true, env: { ...process.env, NODE_ENV: "development", TEST_DIST_DIR: ".next-cps-shots" } });
  await waitForNextServer(base, 180_000);
});
test.afterAll(() => { if (dev?.pid) try { process.kill(-dev.pid, "SIGTERM"); } catch {} });
test("cps checkout shots @checkout", async ({ page }, ti) => {
  test.skip(ti.project.name !== "component-1440");
  test.setTimeout(500_000);
  await page.route("**/photos/classes/**", (route) => { const slug = new URL(route.request().url()).pathname.split("/")[3]!; route.fulfill({ status: 200, contentType: "image/jpeg", body: readFileSync(`${Q}/design/ph/${slug}.jpg`) }); });
  await page.route((u) => u.hostname.includes("cloudflare") || u.hostname.includes("turnstile"), (r) => r.abort());
  await page.route("**/api/quote/reprice", (r) => r.fulfill({ status: 409, contentType: "application/json", body: '{"ok":false,"error":"quote_expired"}' }));
  await page.route("**/api/quote", (r) => { const b = JSON.parse(r.request().postData() ?? "{}"); const pax = Number(b.pax ?? 2); r.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, quote_id: "22222222-2222-4222-8222-222222222222", lock: "v1.l", expires_at: new Date(Date.now() + 1800000).toISOString(), pricing_live: true, classes: CLS.map((c) => ({ slug: c.slug, name: c.name, eligible: pax <= c.pax, ineligible_reason: pax <= c.pax ? null : "pax", effective_max_pax: c.pax, max_bags: c.bags, fixed_route: false, total_rappen: c.t, lines: [], photo_url: `/photos/classes/${c.slug}/x.png` })) }) }); });
  await page.route("**/api/checkout/extras", (r) => r.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, extras: [], vat_rate_bps: 810 }) }));
  await page.route("**/api/checkout/price", (r) => { const b = JSON.parse(r.request().postData() ?? "{}"); const t = CLS.find((c) => c.slug === b.vehicle_class)!.t; r.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, lines: [{ kind: "fare", code: "distance_fare", i18n_key: "price.line.transfer", params: {}, amount_rappen: t }], net_rappen: t, vat_rappen: 0, charged_rappen: t }) }); });
  await page.route("**/api/checkout/me", (r) => r.fulfill({ status: 200, contentType: "application/json", body: '{"signed_in":false}' }));
  await page.route("**/api/checkout/resume**", (r) => r.fulfill({ status: 200, contentType: "application/json", body: '{"state":"none"}' }));
  const q = (pax: number, cls: string) => `from=Zurich%20Airport&fid=dXJuOm1ieHBvaTox&to=Hotel%20Baur%20au%20Lac&tid=dXJuOm1ieHBvaTox2&gs=11111111-1111-4111-8111-111111111111&when=2026-12-15T08:15&pax=${pax}&bags=3&flight=LX318&class=${cls}`;
  for (const w of [390, 768, 1024, 1440]) {
    for (const lang of (w === 1024 ? ["en"] : ["en", "de", "ar"]) as Lang[]) {
      await page.setViewportSize({ width: w, height: w === 390 ? 844 : 1000 });
      await openInLocale(page, base, `/checkout?${q(2, "business")}`, lang);
      await expect(page.locator("[data-co-classes]")).toBeVisible({ timeout: 90_000 });
      await page.waitForFunction(() => [...document.images].every((i) => i.complete), null, { timeout: 20000 });
      await page.waitForTimeout(500);
      await page.locator('[data-co-section="1"], #co-section-class').first().screenshot({ path: `${Q}/screens/built-checkout-${w}-${lang}.png` });
      if (lang === "en") {
        await page.screenshot({ path: `${Q}/screens/built-checkout-${w}-page.png` });
        await openInLocale(page, base, `/checkout?${q(5, "van-luxury")}`, lang);
        await expect(page.locator("[data-co-classes]")).toBeVisible({ timeout: 90_000 });
        await page.waitForTimeout(800);
        await page.locator('[data-co-section="1"], #co-section-class').first().screenshot({ path: `${Q}/screens/built-checkout-${w}-5pax.png` });
      }
      const over = await page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);
      expect(over, `${w} ${lang}`).toBeLessThanOrEqual(0);
    }
  }
});
