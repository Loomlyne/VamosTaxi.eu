async function photos(page: Page) {
  await page.route("**/photos/classes/**", (route) => {
    const slug = new URL(route.request().url()).pathname.split("/")[3]!;
    route.fulfill({ status: 200, contentType: "image/png", body: readFileSync(`${PHD}/${slug}.png`) });
  });
}
function cls(priced: boolean) {
  return CLS.map((c) => ({ slug: c.slug, name: c.name, eligible: true, ineligible_reason: null, effective_max_pax: c.pax, max_bags: c.bags, fixed_route: false, total_rappen: priced ? c.t : null, lines: [], photo_url: `/photos/classes/${c.slug}/x.png` }));
}
async function geo(page: Page) {
  await page.route("**/api/geo/suggest**", async (route) => {
    const q = (new URL(route.request().url()).searchParams.get("q") ?? "").toLowerCase();
    await route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ suggestions: PLACES.filter((s) => s.name.toLowerCase().includes(q)) }) });
  });
  await page.route("**/api/geo/retrieve**", async (route) => {
    const id = new URL(route.request().url()).searchParams.get("mapbox_id");
    const hit = PLACES.find((s) => s.mapbox_id === id);
    await route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, place: hit ? { mapbox_id: id, name: hit.name, isAirport: hit.is_airport } : null }) });
  });
  await page.route("**/api/flight/**", (r) => r.fulfill({ status: 200, contentType: "application/json", body: '{"ok":false}' }));
  await page.route("**/checkout?**", (route) => route.fulfill({ status: 200, contentType: "text/html", body: "<html><body>checkout</body></html>" }));
  await page.route("**/api/quote", async (route: Route) => {
    if (route.request().method() === "GET") return route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, classes: cls(false), fixed_routes: [] }) });
    await route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ok: true, quote_id: "q", lock: "l", expires_at: "2099-01-01T00:00:00Z", pricing_live: true, display_currency: "CHF", classes: cls(true) }) });
  });
}
async function home(page: Page, w: number, h = 900) {

  await waitForMockReady(page);
  const a = page.getByRole("button", { name: /Accept all/i });
  if (await a.isVisible().catch(() => false)) await a.click();
}
async function settle(page: Page) {
  await page.waitForFunction(() => [...document.images].every((i) => i.complete), null, { timeout: 20000 });
  await page.waitForTimeout(600);
}

  await page.keyboard.press("Escape");
}

// ── WebKit and Firefox at the four laptop widths ────────────────────────────────────────
for (const browserName of ["webkit", "firefox", "chromium"] as const) {
  test.describe(`xb ${browserName} @component`, () => {
    for (const w of [1081, 1280, 1360, 1440]) {
      test(`laptop bar ${browserName} ${w}`, async ({}, ti) => {
        test.skip(ti.project.name !== "component-1440");
        test.setTimeout(120_000);
        const br = await { webkit, firefox, chromium }[browserName].launch();
        const page = await br.newPage();
        try {
        await home(page, w);
        const rects = () => page.evaluate(() => {
          const o: Record<string, number[]> = {};
          document.querySelectorAll<HTMLElement>("#book [data-bx]").forEach((e) => { const r = e.getBoundingClientRect(); if (r.width) o[e.getAttribute("data-bx")!] = [Math.round(r.left), Math.round(r.top), Math.round(r.right), Math.round(r.bottom)]; });
          const b = document.querySelector("#book")!.getBoundingClientRect();
          return { o, bar: [Math.round(b.left), Math.round(b.top), Math.round(b.width), Math.round(b.height)], sw: document.documentElement.scrollWidth, cw: document.documentElement.clientWidth };
        });
        const before = await rects();
        const from = page.getByRole("combobox", { name: "From", exact: true });
        await from.fill("Zurich Air");
        const opt = page.getByRole("option").filter({ hasText: "8058 Kloten" }).first();
        await expect(opt).toBeVisible();
        const ob = (await opt.boundingBox())!;
        const topEl = await page.evaluate(([x, y]) => !!document.elementFromPoint(x!, y!)?.closest('[role="option"]'), [ob.x + ob.width / 2, ob.y + ob.height / 2]);
        await opt.click();
        const flight = page.getByRole("textbox", { name: "Flight number" });
        await expect(flight).toBeVisible();
        await page.waitForTimeout(400);
        const focused = await flight.evaluate((e) => e === document.activeElement);
        const after = await rects();
        const names = ["flight", "from", "to", "when", "trav", "cta"].filter((n) => after.o[n]);
        const row = names.map((n) => after.o[n]!);
        let overlap = 0;
        for (let i = 0; i < row.length; i++) for (let j = i + 1; j < row.length; j++) {
          const a = row[i]!, b = row[j]!;
          const ox = Math.min(a[2]!, b[2]!) - Math.max(a[0]!, b[0]!), oy = Math.min(a[3]!, b[3]!) - Math.max(a[1]!, b[1]!);
          if (ox > 1 && oy > 1) overlap++;
        }
        const sameRow = row.every((r) => Math.abs(r[1]! - row[0]![1]!) <= 1);
        const order = sameRow ? row.every((r, i) => i === 0 || r[0]! >= row[i - 1]![2]! - 1) : null;
        const rec = { browser: browserName, w, overlap, listOptionOnTop: topEl, flightFocused: focused, flightBeforeFrom: after.o.flight && after.o.from ? after.o.flight[0]! < after.o.from[0]! || after.o.flight[1]! < after.o.from[1]! : null, sameRow, orderLTR: order, barHeightBefore: before.bar[3], barHeightAfter: after.bar[3], barTopBefore: before.bar[1], barTopAfter: after.bar[1], sideways: after.sw > after.cw, rects: after.o };
        appendFileSync(XB, JSON.stringify(rec) + "\n");
        expect(overlap).toBe(0);
        expect(topEl).toBe(true);
        expect(after.sw).toBeLessThanOrEqual(after.cw);
        } finally { await br.close(); }
      });
    }
  });
}
