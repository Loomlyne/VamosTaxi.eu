catalog.ts contract:
  type Catalogs = { dict: Record<string, { de?: string; fr?: string; ar?: string }>; messages: Record<"en"|"de"|"fr"|"ar", object>; emails: Record<"en"|"de"|"fr"|"ar", object> }
  type Gap = { source: "dict"|"messages"|"emails"; key: string; en: string; langs: ("de"|"fr"|"ar")[] }
  findGaps(catalogs, deny): Gap[]   (a language counts as present when its value is a non-empty string, even if equal to English)
  isDenied(gap, deny): boolean
  applyMessages(jsonText, lang, fills): string (only adds missing leaves; keeps key order and the file's 2-space format and trailing newline)
  applyEmails(jsonText, lang, fills): string (same)
  applyDict(sourceText, fills): string (appends `S["<en>"] = Object.assign({}, S["<en>"], { <lang>: "<text>" });` lines inside the last `var S = DICT.strings` block, with a `/* ── Filled automatically (i18n:fill) */` header line; JS-escaped)
  type Deny = { keys: string[]; prefixes: string[]; texts: string[] }  (texts = exact English strings)
